Arquivo: README.md
# Gelateria Mix — Site oficial (Arcos, MG)
Site com catálogo, carrinho e pedido pelo WhatsApp. Next.js + React + TypeScript + Tailwind CSS + Framer Motion + Lucide.
## Como rodar
    npm install
    npm run dev        # http://localhost:3000
    npm run check      # confere catálogo, preços e mensagem do WhatsApp
    npm run build && npm start
## Onde editar
- Produtos, sabores, tamanhos e preços (em centavos): `lib/catalog.ts`
- Regra do adicional de R$ 3,00 em sundae de 2 sabores: `PRICING_RULES` em `lib/catalog.ts`
- Dados da loja, horários, links e logotipo: `lib/config.ts`
- Fotos da galeria (4 produtos + 2 estabelecimento): `lib/gallery.ts` e `public/images/`
- Avaliações (exatamente 3): `lib/reviews.ts`
- Cálculo de preços: `lib/pricing.ts` | Mensagem do WhatsApp: `lib/whatsapp.ts` | Carrinho: `lib/cart.tsx`
## Publicar na Vercel
1. Suba a pasta para um repositório no GitHub.
2. Na Vercel, importe o repositório (Framework: Next.js, sem configuração extra).
3. Opcional: defina `NEXT_PUBLIC_SITE_URL` com o domínio definitivo para ativar a URL canônica.
## Pendências para a loja
- Enviar o arquivo do logotipo e informar em `BUSINESS.logoSrc` (`lib/config.ts`).
- Enviar 2 fotos do estabelecimento (`lib/gallery.ts`, `STORE_PHOTOS`).
- Preencher nome/data/link das 3 avaliações (`lib/reviews.ts`).
- Confirmar a regra do adicional dos sabores especiais quando o sundae tem 2 sabores especiais.

import { useEffect, useState } from "react";
import { BUSINESS_HOURS } from "@/lib/config";
/** Tabela semanal de horários. O dia de hoje é destacado após a hidratação (evita divergência servidor/cliente). */
export function BusinessHours() {
  const [today, setToday] = useState<number | null>(null);
  useEffect(() => {
    setToday(new Date().getDay());
  }, []);
  return (
    <table className="w-full text-left">
      <caption className="sr-only">Horários de funcionamento da Gelateria Mix</caption>
      <thead className="sr-only">
        <tr>
          <th scope="col">Dia</th>
          <th scope="col">Horário</th>
        </tr>
      </thead>
      <tbody>
        {BUSINESS_HOURS.map((h) => {
          const isToday = today === h.day;
          return (
            <tr
              key={h.day}
              className={`border-b border-line last:border-0 ${isToday ? "bg-magenta/15" : ""}`}
            >
              <th
                scope="row"
                className={`px-3 py-3 text-sm font-semibold ${isToday ? "text-white" : "text-white/80"}`}
              >
                {h.label}
                {isToday && (
                  <span className="ml-2 rounded-full bg-magenta px-2 py-0.5 text-[0.65rem] font-bold uppercase tracking-wider">
                    Hoje
                  </span>
                )}
              </th>
              <td
                className={`px-3 py-3 text-right text-sm tabular-nums ${isToday ? "font-bold text-white" : "text-cream-dim"}`}
              >
                {h.text}
              </td>
            </tr>
          );
        })}
      </tbody>
    </table>
  );
}

// Utilitário local para gerar ids estáveis a partir dos nomes
// ---------------------------------------------------------------------------
function slug(text: string): string {
  return text
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}
function flavors(names: string[], surcharge?: number): Flavor[] {
  return names.map((name) => ({ id: slug(name), name, ...(surcharge ? { surcharge } : {}) }));
}
function sizes(prices: [number, number, number, number]): Size[] {
  const labels = [300, 400, 500, 700];
  return labels.map((ml, i) => ({ id: `${ml}ml`, label: `${ml} ml`, price: prices[i] * 100 }));
}

/** Valida uma seleção e calcula o preço unitário a partir do catálogo. */
export function evaluateSelection(sel: Selection): SelectionStatus {
  const product = getProduct(sel.productId);
  if (!product) return { ok: false, reason: "Produto não encontrado." };
  let size: Size | undefined;
  let base: number;
  if (product.kind === "sized") {
    size = product.sizes?.find((s) => s.id === sel.sizeId);
    if (!size) return { ok: false, reason: "Escolha o tamanho." };
    base = size.price;
  } else {
    base = product.fixedPrice ?? 0;
  }
  if (sel.flavorIds.length !== product.flavorCount) {
    return {
Arquivo (continuação): lib/pricing.ts
      ok: false,
      reason: product.flavorCount === 1 ? "Escolha o sabor." : "Escolha os dois sabores.",
    };
  }
  const flavors: Flavor[] = [];
  for (const id of sel.flavorIds) {
    const f = product.flavors.find((x) => x.id === id);
    if (!f) return { ok: false, reason: "Sabor inválido." };
    flavors.push(f);
  }
  if (
    product.flavorCount === 2 &&
    !product.allowDuplicateFlavors &&
    sel.flavorIds[0] === sel.flavorIds[1]
  ) {
    return { ok: false, reason: "Escolha dois sabores diferentes." };
  }
  const specials = flavors.filter((f) => (f.surcharge ?? 0) > 0);
  let surcharge = 0;
  if (specials.length > 0) {
    surcharge =
      PRICING_RULES.specialSurchargeMode === "per-special-flavor"
        ? specials.reduce((sum, f) => sum + (f.surcharge ?? 0), 0)
        : Math.max(...specials.map((f) => f.surcharge ?? 0));
  }
  return { ok: true, product, size, flavors, base, surcharge, unit: base + surcharge };
}
export function makeKey(sel: Selection): string {
  const product = getProduct(sel.productId);
  const flavorPart =
    product && product.flavorCount === 2
      ? [...sel.flavorIds].sort().join("+")
      : sel.flavorIds.join("+");
  return [sel.productId, sel.sizeId ?? "-", flavorPart].join("|");
}
/** Converte um item do carrinho em linha com preços. Retorna null se o item for inválido. */
export function priceItem(item: CartItem): PricedLine | null {
  const res = evaluateSelection(item);
  if (!res.ok) return null;
  const quantity = Math.max(1, Math.floor(item.quantity));
  return {
    item: { ...item, quantity },
    product: res.product,
    size: res.size,
    flavors: res.flavors,
    base: res.base,
    surcharge: res.surcharge,
    unit: res.unit,
    subtotal: res.unit * quantity,
  };
}
